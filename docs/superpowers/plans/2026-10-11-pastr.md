# pastr Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build pastr, a lean self-hosted Go service that shares spoiler-safe text behind unlisted URLs, with authenticated paste management and a script-free public read page.

**Architecture:** One Go binary serves server-rendered HTML and embedded static assets; one SQLite file in WAL mode holds all state. Packages are layered `config` / `store` / `auth` / `paste` / `sanitize` → `web` → `cmd/pastr`, and only `store` contains SQL.

**Tech Stack:** Go (standard library `net/http`, `html/template`, `log/slog`), `modernc.org/sqlite`, `github.com/microcosm-cc/bluemonday`, `golang.org/x/crypto/argon2`, Trix 2 (vendored), Docker (distroless), GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-10-11-pastr-design.md`

## Global Constraints

- Module path: `github.com/santhoshjanan/pastr`.
- CGO is disabled everywhere (`CGO_ENABLED=0`); no Node toolchain in the repository.
- Only `internal/store` contains SQL.
- The read page (`GET /p/{slug}`) ships no JavaScript except the optional Umami tracker; its body text is never animated or delayed.
- The paste body never appears in preview metadata or in the page served to preview crawlers.
- Expired, burned, deleted, unknown, owner-disabled and not-yours pastes all render the same gone page with HTTP 404.
- No IP addresses or reader data are written to the database or to logs.
- Limits: title 120 characters, body 100 KB (102400 bytes), username 3 to 32 characters of `a-z 0-9 _ -`, password minimum 10 characters, display name 60 characters.
- Slugs are 10 random base62 characters from `crypto/rand`.
- Sessions last 30 days; login throttle is 5 failures per username and per IP in 15 minutes.
- Monospace type is not used anywhere in the UI.
- Copy is plain and friendly: "Show the text", "Link copied", "This one's gone", "That username or password isn't right".
- Timestamps are stored as UTC Unix seconds.
- Commits are small and conventional-style, each ending with the `Co-Authored-By` line configured for the session. Work happens on the branch `feat/initial-build`.
- Scratch files stay outside the repository. Nothing is pushed, and no GitHub repository is created, without the owner's explicit go-ahead.

## Review Focus

1. **Empty-looking body:** Trix submits `<div><br></div>` for an empty editor. Saving must fail with "Write something first." and must not create a paste. (Test in Task 9.)
2. **Hostile title:** a title such as `"><script>alert(1)</script> & 🎬` must appear escaped in the page, in `og:title` and in the crawler page. (Test in Task 8.)
3. **Malformed slug:** `/p/` followed by an over-long, non-base62 or percent-encoded value must return the gone page with 404, never 500. (Test in Task 8.)
4. **Plain-HTTP base URL:** with `PASTR_BASE_URL=http://localhost:8080` the session cookie must not be marked `Secure`, or sign-in silently fails during local use. (Test in Task 7.)
5. **Admin self-lockout:** an admin must not be able to disable their own account, and the last enabled admin must not be disabled. (Test in Task 10.)

```go
// Package config reads pastr's settings from environment variables.
package config

import (
	"errors"
	"fmt"
	"net/url"
	"strings"
)

type Config struct {
	BaseURL        string
	Addr           string
	DBPath         string
	AdminUsername  string
	AdminPassword  string
	TrustedProxy   string
	UmamiScriptURL string
	UmamiWebsiteID string
}

func Load(getenv func(string) string) (Config, error) {
	c := Config{
		BaseURL:        strings.TrimRight(getenv("PASTR_BASE_URL"), "/"),
		Addr:           orDefault(getenv("PASTR_ADDR"), ":8080"),
		DBPath:         orDefault(getenv("PASTR_DB_PATH"), "/data/pastr.db"),
		AdminUsername:  getenv("PASTR_ADMIN_USERNAME"),
		AdminPassword:  getenv("PASTR_ADMIN_PASSWORD"),
		TrustedProxy:   getenv("PASTR_TRUSTED_PROXY"),
		UmamiScriptURL: getenv("PASTR_UMAMI_SCRIPT_URL"),
		UmamiWebsiteID: getenv("PASTR_UMAMI_WEBSITE_ID"),
	}
	if c.BaseURL == "" {
		return c, errors.New("PASTR_BASE_URL is required, for example https://paste.example.com")
	}
	u, err := absoluteURL(c.BaseURL)
	if err != nil {
		return c, fmt.Errorf("PASTR_BASE_URL: %w", err)
	}
	if u.Path != "" {
		return c, errors.New("PASTR_BASE_URL must not contain a path")
	}
	if (c.UmamiScriptURL == "") != (c.UmamiWebsiteID == "") {
		return c, errors.New("set both PASTR_UMAMI_SCRIPT_URL and PASTR_UMAMI_WEBSITE_ID, or neither")
	}
	if c.UmamiScriptURL != "" {
		if _, err := absoluteURL(c.UmamiScriptURL); err != nil {
			return c, fmt.Errorf("PASTR_UMAMI_SCRIPT_URL: %w", err)
		}
	}
	if (c.AdminUsername == "") != (c.AdminPassword == "") {
		return c, errors.New("set both PASTR_ADMIN_USERNAME and PASTR_ADMIN_PASSWORD, or neither")
	}
	return c, nil
}

func (c Config) UmamiEnabled() bool { return c.UmamiScriptURL != "" }

// UmamiOrigin returns scheme and host of the tracker, for the Content Security Policy.
func (c Config) UmamiOrigin() string {
	u, err := url.Parse(c.UmamiScriptURL)
	if err != nil || u.Host == "" {
		return ""
	}
	return u.Scheme + "://" + u.Host
}

func (c Config) SecureCookies() bool { return strings.HasPrefix(c.BaseURL, "https://") }

func absoluteURL(raw string) (*url.URL, error) {
	u, err := url.Parse(raw)
	if err != nil {
		return nil, err
	}
	if (u.Scheme != "http" && u.Scheme != "https") || u.Host == "" {
		return nil, errors.New("must be an absolute http or https URL")
	}
	return u, nil
}

func orDefault(v, d string) string {
	if v == "" {
		return d
	}
	return v
}
```

```go
// Package store is pastr's SQLite persistence layer and the only package containing SQL.
package store

import (
	"context"
	"database/sql"
	"embed"
	"errors"
	"fmt"
	"io/fs"
	"sort"
	"strings"

	_ "modernc.org/sqlite"
)

var (
	ErrNotFound = errors.New("store: not found")
	ErrConflict = errors.New("store: conflict")
)

//go:embed migrations/*.sql
var migrationFS embed.FS

type Store struct{ db *sql.DB }

func Open(path string) (*Store, error) {
	dsn := "file:" + path + "?_pragma=journal_mode(WAL)&_pragma=foreign_keys(1)&_pragma=busy_timeout(5000)&_txlock=immediate"
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	s := &Store{db: db}
	if err := s.migrate(context.Background()); err != nil {
		db.Close()
		return nil, fmt.Errorf("migrate: %w", err)
	}
	return s, nil
}

func (s *Store) Close() error                   { return s.db.Close() }
func (s *Store) Ping(ctx context.Context) error { return s.db.PingContext(ctx) }

func (s *Store) migrate(ctx context.Context) error {
	if _, err := s.db.ExecContext(ctx, `CREATE TABLE IF NOT EXISTS schema_migrations (name TEXT PRIMARY KEY)`); err != nil {
		return err
	}
	names, err := fs.Glob(migrationFS, "migrations/*.sql")
	if err != nil {
		return err
	}
	sort.Strings(names)
	for _, name := range names {
		var n int
		if err := s.db.QueryRowContext(ctx, `SELECT COUNT(*) FROM schema_migrations WHERE name = ?`, name).Scan(&n); err != nil {
			return err
		}
		if n > 0 {
			continue
		}
		body, err := migrationFS.ReadFile(name)
		if err != nil {
			return err
		}
		tx, err := s.db.BeginTx(ctx, nil)
		if err != nil {
			return err
		}
		if _, err := tx.ExecContext(ctx, string(body)); err != nil {
			tx.Rollback()
			return fmt.Errorf("%s: %w", name, err)
		}
		if _, err := tx.ExecContext(ctx, `INSERT INTO schema_migrations (name) VALUES (?)`, name); err != nil {
			tx.Rollback()
			return err
		}
		if err := tx.Commit(); err != nil {
			return err
		}
	}
	return nil
}

// mapErr translates driver errors into the package's sentinel errors.
func mapErr(err error) error {
	switch {
	case err == nil:
		return nil
	case errors.Is(err, sql.ErrNoRows):
		return ErrNotFound
	case strings.Contains(err.Error(), "UNIQUE constraint failed"):
		return ErrConflict
	}
	return err
}
```

```go
package store

import (
	"context"
	"time"
)

type User struct {
	ID           int64
	Username     string
	PasswordHash string
	DisplayName  string
	DisplayLink  string
	IsAdmin      bool
	Disabled     bool
	CreatedAt    time.Time
}

type Session struct {
	TokenHash string
	UserID    int64
	CSRFToken string
	ExpiresAt time.Time
	CreatedAt time.Time
}

const userCols = `id, username, password_hash, display_name, display_link, is_admin, disabled, created_at`

type scanner interface{ Scan(dest ...any) error }

func scanUser(row scanner) (User, error) {
	var u User
	var created int64
	err := row.Scan(&u.ID, &u.Username, &u.PasswordHash, &u.DisplayName, &u.DisplayLink, &u.IsAdmin, &u.Disabled, &created)
	u.CreatedAt = time.Unix(created, 0).UTC()
	return u, mapErr(err)
}

func (s *Store) CreateUser(ctx context.Context, username, hash string, isAdmin bool, now time.Time) (User, error) {
	res, err := s.db.ExecContext(ctx,
		`INSERT INTO users (username, password_hash, is_admin, created_at) VALUES (?, ?, ?, ?)`,
		username, hash, isAdmin, now.Unix())
	if err != nil {
		return User{}, mapErr(err)
	}
	id, err := res.LastInsertId()
	if err != nil {
		return User{}, err
	}
	return s.UserByID(ctx, id)
}

func (s *Store) UserByUsername(ctx context.Context, username string) (User, error) {
	return scanUser(s.db.QueryRowContext(ctx, `SELECT `+userCols+` FROM users WHERE username = ?`, username))
}

func (s *Store) UserByID(ctx context.Context, id int64) (User, error) {
```

```go
package store

import (
	"context"
	"database/sql"
	"time"
)

type Paste struct {
	Slug       string
	UserID     int64
	Title      string
	BodyHTML   string
	RevealMode string
	ExpiresAt  *time.Time
	MaxViews   *int
	ViewCount  int
	CreatedAt  time.Time
	UpdatedAt  time.Time
}

// PasteView is a paste as the public sees it, with the author's optional byline.
type PasteView struct {
	Paste
	AuthorName string
	AuthorLink string
}

const pasteCols = `p.slug, p.user_id, p.title, p.body_html, p.reveal_mode, p.expires_at, p.max_views, p.view_count, p.created_at, p.updated_at`

func scanPaste(row scanner, extra ...any) (Paste, error) {
	var p Paste
	var exp, maxViews sql.NullInt64
	var created, updated int64
	dest := append([]any{&p.Slug, &p.UserID, &p.Title, &p.BodyHTML, &p.RevealMode, &exp, &maxViews, &p.ViewCount, &created, &updated}, extra...)
	if err := row.Scan(dest...); err != nil {
		return p, mapErr(err)
	}
	if exp.Valid {
		t := time.Unix(exp.Int64, 0).UTC()
		p.ExpiresAt = &t
	}
	if maxViews.Valid {
		n := int(maxViews.Int64)
		p.MaxViews = &n
	}
	p.CreatedAt, p.UpdatedAt = time.Unix(created, 0).UTC(), time.Unix(updated, 0).UTC()
	return p, nil
}

func nullTime(t *time.Time) any {
	if t == nil {
		return nil
	}
	return t.Unix()
}

func nullInt(n *int) any {
	if n == nil {
		return nil
	}
	return *n
}

func (s *Store) CreatePaste(ctx context.Context, p Paste) error {
	_, err := s.db.ExecContext(ctx,
		`INSERT INTO pastes (slug, user_id, title, body_html, reveal_mode, expires_at, max_views, created_at, updated_at)
		 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		p.Slug, p.UserID, p.Title, p.BodyHTML, p.RevealMode, nullTime(p.ExpiresAt), nullInt(p.MaxViews), p.CreatedAt.Unix(), p.UpdatedAt.Unix())
	return mapErr(err)
}

func (s *Store) PasteBySlug(ctx context.Context, slug string) (Paste, error) {
	return scanPaste(s.db.QueryRowContext(ctx, `SELECT `+pasteCols+` FROM pastes p WHERE p.slug = ?`, slug))
}

// ReadPaste is the public lookup. Anything a reader must not see is reported as ErrNotFound.
func (s *Store) ReadPaste(ctx context.Context, slug string, now time.Time) (PasteView, error) {
	var v PasteView
	p, err := scanPaste(s.db.QueryRowContext(ctx,
		`SELECT `+pasteCols+`, u.display_name, u.display_link
		 FROM pastes p JOIN users u ON u.id = p.user_id
		 WHERE p.slug = ? AND u.disabled = 0
		   AND (p.expires_at IS NULL OR p.expires_at > ?)
		   AND (p.max_views IS NULL OR p.view_count < p.max_views)`,
		slug, now.Unix()), &v.AuthorName, &v.AuthorLink)
	v.Paste = p
	return v, err
}

func (s *Store) ListPastesByUser(ctx context.Context, userID int64, now time.Time) ([]Paste, error) {
	rows, err := s.db.QueryContext(ctx,
		`SELECT `+pasteCols+` FROM pastes p
		 WHERE p.user_id = ? AND (p.expires_at IS NULL OR p.expires_at > ?)
		 ORDER BY p.created_at DESC, p.slug`, userID, now.Unix())
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []Paste
	for rows.Next() {
		p, err := scanPaste(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, p)
	}
	return out, rows.Err()
}

func (s *Store) UpdatePaste(ctx context.Context, p Paste) error {
	res, err := s.db.ExecContext(ctx,
		`UPDATE pastes SET title = ?, body_html = ?, reveal_mode = ?, expires_at = ?, max_views = ?, updated_at = ?
		 WHERE slug = ? AND user_id = ?`,
		p.Title, p.BodyHTML, p.RevealMode, nullTime(p.ExpiresAt), nullInt(p.MaxViews), p.UpdatedAt.Unix(), p.Slug, p.UserID)
	return affected(res, err)
}

func (s *Store) DeletePaste(ctx context.Context, slug string, userID int64) error {
	res, err := s.db.ExecContext(ctx, `DELETE FROM pastes WHERE slug = ? AND user_id = ?`, slug, userID)
	return affected(res, err)
}

// CountView records one view. It reports false when the view limit is already used up.
// The check and the increment are one statement, so concurrent readers cannot overshoot.
func (s *Store) CountView(ctx context.Context, slug string) (bool, error) {
	res, err := s.db.ExecContext(ctx,
		`UPDATE pastes SET view_count = view_count + 1
		 WHERE slug = ? AND (max_views IS NULL OR view_count < max_views)`, slug)
	if err != nil {
		return false, err
	}
	n, err := res.RowsAffected()
	return n == 1, err
}

func (s *Store) DeleteIfBurned(ctx context.Context, slug string) error {
	_, err := s.db.ExecContext(ctx,
		`DELETE FROM pastes WHERE slug = ? AND max_views IS NOT NULL AND view_count >= max_views`, slug)
	return err
}

func (s *Store) DeleteExpiredPastes(ctx context.Context, now time.Time) (int64, error) {
	res, err := s.db.ExecContext(ctx, `DELETE FROM pastes WHERE expires_at IS NOT NULL AND expires_at <= ?`, now.Unix())
	if err != nil {
		return 0, err
	}
	return res.RowsAffected()
}

func affected(res sql.Result, err error) error {
	if err != nil {
		return mapErr(err)
	}
	n, err := res.RowsAffected()
	if err != nil {
		return err
	}
	if n == 0 {
		return ErrNotFound
	}
	return nil
}
```

```go
// Package auth handles passwords, sessions and sign-in throttling.
package auth

import (
	"crypto/rand"
	"crypto/subtle"
	"encoding/base64"
	"errors"
	"fmt"
	"regexp"
	"strings"

	"golang.org/x/crypto/argon2"
)

const (
	argonTime    = 3
	argonMemory  = 64 * 1024 // KiB
	argonThreads = 2
	argonKeyLen  = 32
	argonSaltLen = 16

	MinPasswordLen = 10
	maxPasswordLen = 1024
)

var usernameRE = regexp.MustCompile(`^[a-z0-9_-]{3,32}$`)

func HashPassword(pw string) (string, error) {
	salt := make([]byte, argonSaltLen)
	if _, err := rand.Read(salt); err != nil {
		return "", err
	}
	key := argon2.IDKey([]byte(pw), salt, argonTime, argonMemory, argonThreads, argonKeyLen)
	return fmt.Sprintf("$argon2id$v=%d$m=%d,t=%d,p=%d$%s$%s", argon2.Version, argonMemory, argonTime, argonThreads,
		base64.RawStdEncoding.EncodeToString(salt), base64.RawStdEncoding.EncodeToString(key)), nil
}

func VerifyPassword(encoded, pw string) bool {
	parts := strings.Split(encoded, "$")
	if len(parts) != 6 || parts[1] != "argon2id" {
		return false
	}
	var version int
	var memory, time uint32
	var threads uint8
	if _, err := fmt.Sscanf(parts[2], "v=%d", &version); err != nil || version != argon2.Version {
		return false
	}
	if _, err := fmt.Sscanf(parts[3], "m=%d,t=%d,p=%d", &memory, &time, &threads); err != nil {
		return false
	}
	// Refuse parameters far above ours, so a corrupted row cannot exhaust memory.
	if memory > 4*argonMemory || time > 10 || threads == 0 {
		return false
	}
	salt, err := base64.RawStdEncoding.DecodeString(parts[4])
	if err != nil {
		return false
	}
	want, err := base64.RawStdEncoding.DecodeString(parts[5])
	if err != nil || len(want) == 0 {
		return false
	}
	got := argon2.IDKey([]byte(pw), salt, time, memory, threads, uint32(len(want)))
	return subtle.ConstantTimeCompare(got, want) == 1
}

// BurnTime does the work of a verification and discards it, so that signing in
// with an unknown username takes as long as signing in with a wrong password.
func BurnTime(pw string) {
	argon2.IDKey([]byte(pw), make([]byte, argonSaltLen), argonTime, argonMemory, argonThreads, argonKeyLen)
}

func ValidatePassword(pw string) error {
	if len(pw) < MinPasswordLen {
		return fmt.Errorf("Use at least %d characters.", MinPasswordLen)
	}
	if len(pw) > maxPasswordLen {
		return errors.New("That password is too long.")
	}
	return nil
}

// ValidateUsername trims and lower-cases the name, then checks its shape.
func ValidateUsername(raw string) (string, error) {
	u := strings.ToLower(strings.TrimSpace(raw))
	if !usernameRE.MatchString(u) {
		return "", errors.New("Usernames are 3 to 32 characters: letters, numbers, dashes and underscores.")
	}
	return u, nil
}
```

```go
// NewToken returns 32 random bytes, URL-safe encoded.
func NewToken() (string, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b), nil
}

// HashToken is what gets stored, so a copy of the database cannot be replayed as a cookie.
func HashToken(token string) string {
	sum := sha256.Sum256([]byte(token))
	return hex.EncodeToString(sum[:])
}

type Sessions struct {
	Store *store.Store
	Now   func() time.Time
}

func (s Sessions) Start(ctx context.Context, userID int64) (string, error) {
	token, err := NewToken()
	if err != nil {
		return "", err
	}
	csrf, err := NewToken()
	if err != nil {
		return "", err
	}
	now := s.Now()
	err = s.Store.CreateSession(ctx, store.Session{
		TokenHash: HashToken(token), UserID: userID, CSRFToken: csrf,
		ExpiresAt: now.Add(SessionTTL), CreatedAt: now,
	})
	return token, err
}

// Lookup returns store.ErrNotFound for unknown, expired and disabled-user sessions alike.
func (s Sessions) Lookup(ctx context.Context, token string) (store.Session, store.User, error) {
	sess, err := s.Store.SessionByTokenHash(ctx, HashToken(token), s.Now())
	if err != nil {
		return store.Session{}, store.User{}, err
	}
	u, err := s.Store.UserByID(ctx, sess.UserID)
	if err != nil {
		return store.Session{}, store.User{}, err
	}
	if u.Disabled {
		return store.Session{}, store.User{}, store.ErrNotFound
	}
	return sess, u, nil
}

func (s Sessions) End(ctx context.Context, token string) error {
	return s.Store.DeleteSession(ctx, HashToken(token))
```

```go
package auth

import (
	"sync"
	"time"
)

// Throttle counts recent failures per key in memory. Nothing is persisted,
// so keys that contain IP addresses vanish on restart.
type Throttle struct {
	mu     sync.Mutex
	max    int
	window time.Duration
	now    func() time.Time
	fails  map[string][]time.Time
}

func NewThrottle(max int, window time.Duration, now func() time.Time) *Throttle {
	return &Throttle{max: max, window: window, now: now, fails: map[string][]time.Time{}}
}

func (t *Throttle) Allowed(key string) bool {
	t.mu.Lock()
	defer t.mu.Unlock()
	return len(t.recent(key)) < t.max
}

func (t *Throttle) Fail(key string) {
	t.mu.Lock()
	defer t.mu.Unlock()
	t.fails[key] = append(t.recent(key), t.now())
	if len(t.fails) > 10_000 { // bound memory under a flood of distinct keys
		for k := range t.fails {
			if len(t.recent(k)) == 0 {
				delete(t.fails, k)
			}
		}
	}
}

func (t *Throttle) Reset(key string) {
	t.mu.Lock()
	defer t.mu.Unlock()
	delete(t.fails, key)
}

func (t *Throttle) recent(key string) []time.Time {
	cutoff := t.now().Add(-t.window)
	kept := t.fails[key][:0]
	for _, at := range t.fails[key] {
		if at.After(cutoff) {
			kept = append(kept, at)
		}
	}
	if len(kept) == 0 {
		delete(t.fails, key)
		return nil
	}
	t.fails[key] = kept
	return kept
}
```

```go
// Package sanitize reduces author-supplied HTML to pastr's small formatting allowlist.
package sanitize

import (
	"regexp"
	"strings"

	"github.com/microcosm-cc/bluemonday"
)

var (
	policy = newPolicy()
	strict = bluemonday.StrictPolicy()
	// The editor's only heading is <h1>; the page title already owns that level.
	h1Open  = regexp.MustCompile(`(?i)<h1(\s[^>]*)?>`)
	h1Close = regexp.MustCompile(`(?i)</h1\s*>`)
)

func newPolicy() *bluemonday.Policy {
	p := bluemonday.NewPolicy()
	p.AllowElements("div", "p", "br", "strong", "b", "em", "i", "del", "s", "h2", "blockquote", "ul", "ol", "li")
	p.AllowAttrs("href").OnElements("a")
	p.AllowURLSchemes("http", "https", "mailto")
	p.RequireParseableURLs(true)
	p.RequireNoFollowOnLinks(true)
	p.RequireNoReferrerOnLinks(true)
	p.AddTargetBlankToFullyQualifiedLinks(true) // also adds rel="noopener"
	return p
}

// HTML returns input with everything outside the allowlist removed.
// It is applied once, when a paste is saved.
func HTML(input string) string {
	s := h1Open.ReplaceAllString(input, "<h2>")
	s = h1Close.ReplaceAllString(s, "</h2>")
	s = policy.Sanitize(s)
	// Added after sanitising, so authors cannot supply tracking attributes of their own.
	return strings.ReplaceAll(s, `<a href=`, `<a data-umami-event="outbound-link" href=`)
}

// IsBlank reports whether sanitised HTML contains no visible text.
func IsBlank(html string) bool {
	text := strict.Sanitize(html)
	text = strings.ReplaceAll(text, "&nbsp;", " ")
	text = strings.ReplaceAll(text, " ", " ")
	return strings.TrimSpace(text) == ""
}
```

```go
// Package paste holds the rules for pastes that do not need the database.
package paste

import (
	"crypto/rand"
	"errors"
	"fmt"
	"hash/fnv"
	"strconv"
	"strings"
	"time"
	"unicode/utf8"
)

const (
	SlugLen       = 10
	MaxTitleChars = 120
	MaxBodyBytes  = 100 * 1024
	maxViewLimit  = 1_000_000

	alphabet = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
)

var colours = [...]string{"red", "blue", "green", "orange"}

// NewSlug draws 10 base62 characters from crypto/rand without modulo bias.
func NewSlug() (string, error) {
	out := make([]byte, 0, SlugLen)
	buf := make([]byte, SlugLen*2)
	for len(out) < SlugLen {
		if _, err := rand.Read(buf); err != nil {
			return "", err
		}
		for _, b := range buf {
			if b < 248 && len(out) < SlugLen { // 248 = 62 * 4
				out = append(out, alphabet[int(b)%62])
			}
		}
	}
	return string(out), nil
}

func ValidSlug(s string) bool {
	if len(s) != SlugLen {
		return false
	}
	for i := 0; i < len(s); i++ {
		if !strings.ContainsRune(alphabet, rune(s[i])) {
			return false
		}
	}
	return true
}

// SignalColour gives each paste one of four colours that never changes.
func SignalColour(slug string) string {
	h := fnv.New32a()
	h.Write([]byte(slug))
	return colours[h.Sum32()%uint32(len(colours))]
}

// ValidateTitle collapses whitespace and enforces the length limit in characters.
func ValidateTitle(raw string) (string, error) {
	title := strings.Join(strings.Fields(raw), " ")
	if title == "" {
		return "", errors.New("Give it a title. It's what people see before they open it.")
	}
	if utf8.RuneCountInString(title) > MaxTitleChars {
		return "", fmt.Errorf("Keep the title under %d characters.", MaxTitleChars)
	}
	return title, nil
}

func ExpiresAt(choice string, now time.Time) (*time.Time, error) {
	var d time.Duration
	switch choice {
	case "never":
		return nil, nil
	case "1d":
		d = 24 * time.Hour
	case "1w":
		d = 7 * 24 * time.Hour
	case "1m":
		d = 30 * 24 * time.Hour
	default:
		return nil, errors.New("Pick how long this should last.")
	}
	t := now.Add(d).UTC()
	return &t, nil
}

func ParseMaxViews(raw string) (*int, error) {
	raw = strings.TrimSpace(raw)
	if raw == "" {
		return nil, nil
	}
	n, err := strconv.Atoi(raw)
	if err != nil || n < 1 || n > maxViewLimit {
		return nil, errors.New("The view limit is a whole number, 1 or more. Leave it blank for no limit.")
	}
	return &n, nil
}

// TimeLeft returns a short phrase such as "gone in 3 days", or "" when there is no expiry.
func TimeLeft(expires *time.Time, now time.Time) string {
	if expires == nil {
		return ""
	}
	d := expires.Sub(now)
	switch {
	case d < time.Minute:
		return "gone in under a minute"
	case d < time.Hour:
		return "gone in " + plural(int(d/time.Minute), "minute")
	case d < 24*time.Hour:
		return "gone in " + plural(int(d/time.Hour), "hour")
	}
	return "gone in " + plural(int(d/(24*time.Hour)), "day")
}

func plural(n int, unit string) string {
	if n == 1 {
		return "1 " + unit
	}
	return strconv.Itoa(n) + " " + unit + "s"
}
```

```go
func hashFS(fsys fs.FS) (string, error) {
	h := sha256.New()
	err := fs.WalkDir(fsys, ".", func(p string, d fs.DirEntry, err error) error {
		if err != nil || d.IsDir() {
			return err
		}
		b, err := fs.ReadFile(fsys, p)
		if err != nil {
			return err
		}
		h.Write([]byte(p))
		h.Write(b)
		return nil
	})
	return hex.EncodeToString(h.Sum(nil))[:12], err
}
```

```go
// recoverer tags each request with an ID and turns panics into the plain error page.
func (s *Server) recoverer(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		b := make([]byte, 8)
		rand.Read(b)
		id := hex.EncodeToString(b)
		r = r.WithContext(context.WithValue(r.Context(), keyRequestID, id))
		defer func() {
			if v := recover(); v != nil {
				s.log.Error("panic", "request_id", id, "path", r.URL.Path, "panic", v)
				s.render(w, r, http.StatusInternalServerError, "error.html", pageData{Title: "Something broke"})
			}
		}()
		next.ServeHTTP(w, r)
	})
}

func (s *Server) securityHeaders(next http.Handler) http.Handler {
	script, connect := "'self'", "'self'"
	if o := s.cfg.UmamiOrigin(); o != "" {
		script += " " + o
		connect += " " + o
	}
	csp := "default-src 'none'; style-src 'self'; script-src " + script + "; connect-src " + connect +
		"; img-src 'self' data:; form-action 'self'; base-uri 'none'; frame-ancestors 'none'"
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		h := w.Header()
		h.Set("Content-Security-Policy", csp)
		h.Set("X-Content-Type-Options", "nosniff")
		h.Set("Referrer-Policy", "same-origin")
		if s.cfg.SecureCookies() {
			h.Set("Strict-Transport-Security", "max-age=31536000")
		}
		next.ServeHTTP(w, r)
	})
}

func (s *Server) limitBody(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		r.Body = http.MaxBytesReader(w, r.Body, maxRequestBody)
		next.ServeHTTP(w, r)
	})
}

func (s *Server) loadSession(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if ck, err := r.Cookie(cookieName); err == nil && ck.Value != "" {
			if sess, user, err := s.sessions.Lookup(r.Context(), ck.Value); err == nil {
				ctx := context.WithValue(r.Context(), keyUser, &user)
				ctx = context.WithValue(ctx, keySession, &sess)
				r = r.WithContext(ctx)
				w.Header().Set("Cache-Control", "no-store")
			}
		}
		next.ServeHTTP(w, r)
	})
}

// protect rejects state-changing requests that do not come from our own pages:
// the Origin header must match, and a signed-in request must carry the session's CSRF token.
func (s *Server) protect(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method == http.MethodGet || r.Method == http.MethodHead || r.Method == http.MethodOptions {
			next.ServeHTTP(w, r)
			return
		}
		if r.Header.Get("Origin") != s.baseOrigin {
			http.Error(w, "Forbidden", http.StatusForbidden)
			return
		}
		if sess := currentSession(r); sess != nil {
			if err := r.ParseForm(); err != nil {
				http.Error(w, "Request too large", http.StatusRequestEntityTooLarge)
				return
			}
			if subtle.ConstantTimeCompare([]byte(r.PostFormValue("csrf")), []byte(sess.CSRFToken)) != 1 {
				http.Error(w, "Forbidden", http.StatusForbidden)
				return
			}
		}
		next.ServeHTTP(w, r)
	})
}

func (s *Server) requireUser(next http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		if currentUser(r) == nil {
			http.Redirect(w, r, "/login", http.StatusSeeOther)
			return
		}
		next(w, r)
	}
}

func (s *Server) requireAdmin(next http.HandlerFunc) http.HandlerFunc {
	return s.requireUser(func(w http.ResponseWriter, r *http.Request) {
		if !currentUser(r).IsAdmin {
			s.gone(w, r)
			return
		}
		next(w, r)
	})
}

// clientIP is used only as an in-memory throttle key. The forwarded header is
// believed only when the request arrives from the configured proxy.
func (s *Server) clientIP(r *http.Request) string {
	host, _, err := net.SplitHostPort(r.RemoteAddr)
	if err != nil {
		host = r.RemoteAddr
	}
	if s.cfg.TrustedProxy != "" && host == s.cfg.TrustedProxy {
		parts := strings.Split(r.Header.Get("X-Forwarded-For"), ",")
		if last := strings.TrimSpace(parts[len(parts)-1]); last != "" {
			return last
		}
	}
	return host
}

func (s *Server) setSessionCookie(w http.ResponseWriter, token string, maxAge int) {
	http.SetCookie(w, &http.Cookie{
		Name: cookieName, Value: token, Path: "/", MaxAge: maxAge,
		HttpOnly: true, Secure: s.cfg.SecureCookies(), SameSite: http.SameSiteLaxMode,
	})
}
```

```go
	Error      string
	Notice     string

	"gone.html", "error.html", "login.html", "signup.html", "account.html",

func loadPages(s *Server) (map[string]*template.Template, error) {
	funcs := template.FuncMap{"asset": s.asset}
	pages := map[string]*template.Template{}
	for _, name := range pageNames {
		t, err := template.New("base.html").Funcs(funcs).ParseFS(templateFS, "templates/base.html", "templates/"+name)
		if err != nil {
			return nil, err
		}
		pages[name] = t
	}
	return pages, nil
}

func (s *Server) render(w http.ResponseWriter, r *http.Request, status int, page string, d pageData) {
	d.User = currentUser(r)
	if sess := currentSession(r); sess != nil {
		d.CSRF = sess.CSRFToken
	}
	if d.Colour == "" {
		d.Colour = "red"
	}
	d.UmamiScript, d.UmamiID = s.cfg.UmamiScriptURL, s.cfg.UmamiWebsiteID

	var buf bytes.Buffer
	if err := s.pages[page].ExecuteTemplate(&buf, "base", d); err != nil {
		s.log.Error("render", "page", page, "err", err)
		http.Error(w, "Something broke on our side.", http.StatusInternalServerError)
		return
	}
	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	w.WriteHeader(status)
	buf.WriteTo(w)
}

// gone is the one response for anything a visitor may not see or that does not exist.
func (s *Server) gone(w http.ResponseWriter, r *http.Request) {
	w.Header().Set("X-Robots-Tag", "noindex, nofollow")
	s.render(w, r, http.StatusNotFound, "gone.html", pageData{Title: "This one's gone", Track: "gone"})
}

func (s *Server) fail(w http.ResponseWriter, r *http.Request, err error) {
	if errors.Is(err, store.ErrNotFound) {
		s.gone(w, r)
		return
	}
	id, _ := r.Context().Value(keyRequestID).(string)
	s.log.Error("request failed", "request_id", id, "path", r.URL.Path, "err", err)
	s.render(w, r, http.StatusInternalServerError, "error.html", pageData{Title: "Something broke"})
```

```go
const msgBadLogin = "That username or password isn't right."

func (s *Server) handleLoginForm(w http.ResponseWriter, r *http.Request) {
	if currentUser(r) != nil {
		http.Redirect(w, r, "/", http.StatusSeeOther)
		return
	}
	s.render(w, r, http.StatusOK, "login.html", pageData{Title: "Sign in", Data: ""})
}

func (s *Server) handleLogin(w http.ResponseWriter, r *http.Request) {
	username := strings.ToLower(strings.TrimSpace(r.PostFormValue("username")))
	password := r.PostFormValue("password")
	userKey, ipKey := "u:"+username, "ip:"+s.clientIP(r)

	if !s.logins.Allowed(userKey) || !s.logins.Allowed(ipKey) {
		s.render(w, r, http.StatusTooManyRequests, "login.html",
			pageData{Title: "Sign in", Data: username, Error: "Too many tries. Wait a few minutes, then try again."})
		return
	}
	u, err := s.store.UserByUsername(r.Context(), username)
	ok := false
	switch {
	case err == nil:
		ok = auth.VerifyPassword(u.PasswordHash, password) && !u.Disabled
	case errors.Is(err, store.ErrNotFound):
		auth.BurnTime(password) // same cost as a real check
	default:
		s.fail(w, r, err)
		return
	}
	if !ok {
		s.logins.Fail(userKey)
		s.logins.Fail(ipKey)
		s.render(w, r, http.StatusUnauthorized, "login.html", pageData{Title: "Sign in", Data: username, Error: msgBadLogin})
		return
	}
	s.logins.Reset(userKey)
	s.startSession(w, r, u.ID, "/?e=login")
}

func (s *Server) startSession(w http.ResponseWriter, r *http.Request, userID int64, to string) {
	token, err := s.sessions.Start(r.Context(), userID)
	if err != nil {
		s.fail(w, r, err)
		return
	}
	s.setSessionCookie(w, token, int(auth.SessionTTL.Seconds()))
	http.Redirect(w, r, to, http.StatusSeeOther)
}

func (s *Server) handleLogout(w http.ResponseWriter, r *http.Request) {
	if ck, err := r.Cookie(cookieName); err == nil {
		s.sessions.End(r.Context(), ck.Value)
	}
	s.setSessionCookie(w, "", -1)
	http.Redirect(w, r, "/login", http.StatusSeeOther)
}

func (s *Server) signupsOpen(w http.ResponseWriter, r *http.Request) bool {
	on, err := s.store.SignupsEnabled(r.Context())
	if err != nil {
		s.fail(w, r, err)
		return false
	}
	if !on {
		s.gone(w, r)
	}
	return on
}

func (s *Server) handleSignupForm(w http.ResponseWriter, r *http.Request) {

		return
	}

func (s *Server) handleSignup(w http.ResponseWriter, r *http.Request) {
	if !s.signupsOpen(w, r) {
		return
	}
	raw := r.PostFormValue("username")
	reject := func(status int, msg string) {
		s.render(w, r, status, "signup.html", pageData{Title: "Create an account", Data: raw, Error: msg})
	}
	ipKey := "ip:" + s.clientIP(r)
	if !s.signups.Allowed(ipKey) {
		reject(http.StatusTooManyRequests, "Too many sign-ups from here. Try again later.")
		return
	}
	username, err := auth.ValidateUsername(raw)
	if err != nil {
		reject(http.StatusUnprocessableEntity, err.Error())
		return
	}
	password := r.PostFormValue("password")
	if err := auth.ValidatePassword(password); err != nil {
		reject(http.StatusUnprocessableEntity, err.Error())
		return
	}
	hash, err := auth.HashPassword(password)
	if err != nil {
		s.fail(w, r, err)
		return
	}
	s.signups.Fail(ipKey) // every attempt that reaches the database counts
	u, err := s.store.CreateUser(r.Context(), username, hash, false, s.now())
	if errors.Is(err, store.ErrConflict) {
		reject(http.StatusUnprocessableEntity, "That username is taken.")
		return
	}
	if err != nil {
		s.fail(w, r, err)
		return
	}
	s.startSession(w, r, u.ID, "/?e=signup")
}

func (s *Server) handleAccountForm(w http.ResponseWriter, r *http.Request) {
	d := pageData{Title: "Account"}
	switch r.URL.Query().Get("saved") {
	case "profile":
		d.Notice = "Byline saved."
	case "password":
		d.Notice = "Password changed."
	}
	s.render(w, r, http.StatusOK, "account.html", d)
}

func (s *Server) handleAccount(w http.ResponseWriter, r *http.Request) {
	u := currentUser(r)
	reject := func(msg string) {
		s.render(w, r, http.StatusUnprocessableEntity, "account.html", pageData{Title: "Account", Error: msg})
	}
	switch r.PostFormValue("action") {
	case "profile":
		name := strings.Join(strings.Fields(r.PostFormValue("display_name")), " ")
		link := strings.TrimSpace(r.PostFormValue("display_link"))
		if utf8.RuneCountInString(name) > 60 {
			reject("Keep the display name under 60 characters.")
			return
		}
		if link != "" {
			lu, err := url.Parse(link)
			if err != nil || (lu.Scheme != "http" && lu.Scheme != "https") || lu.Host == "" || len(link) > 300 {
				reject("The link needs to start with https:// or http://.")
				return
			}
		}
		if err := s.store.UpdateProfile(r.Context(), u.ID, name, link); err != nil {
			s.fail(w, r, err)
			return
		}
		http.Redirect(w, r, "/account?saved=profile", http.StatusSeeOther)
	case "password":
		if !auth.VerifyPassword(u.PasswordHash, r.PostFormValue("current")) {
			reject("Your current password isn't right.")
			return
		}
		if err := auth.ValidatePassword(r.PostFormValue("new")); err != nil {
			reject(err.Error())
			return
		}
		hash, err := auth.HashPassword(r.PostFormValue("new"))
		if err != nil {
			s.fail(w, r, err)
			return
		}
		if err := s.store.UpdatePassword(r.Context(), u.ID, hash); err != nil {
			s.fail(w, r, err)
			return
		}
		http.Redirect(w, r, "/account?saved=password", http.StatusSeeOther)
	default:
		s.gone(w, r)
	}
}
```

```go
import (
	"html/template"
	"net/http"

	"github.com/santhoshjanan/pastr/internal/paste"

	Gated    bool
	TimeLeft string
	Date     string
	URL      string
}

func (s *Server) handleRead(w http.ResponseWriter, r *http.Request) {
	slug := r.PathValue("slug")
	if !paste.ValidSlug(slug) {
		s.gone(w, r)
		return
	}
	v, err := s.store.ReadPaste(r.Context(), slug, s.now())
	if err != nil {
		s.fail(w, r, err)
		return
	}
	h := w.Header()
	h.Set("X-Robots-Tag", "noindex, nofollow")
	h.Set("Referrer-Policy", "no-referrer")
	h.Set("Cache-Control", "no-store")

	d := pageData{Title: v.Title, Colour: paste.SignalColour(slug), Data: readData{
		Paste: v, URL: s.cfg.BaseURL + "/p/" + slug,
	}}
	// Link-preview fetchers get the title only, and are not readers.
	if paste.IsPreviewCrawler(r.UserAgent()) {
		s.render(w, r, http.StatusOK, "preview.html", d)
		return
	}

	owner := currentUser(r) != nil && currentUser(r).ID == v.UserID
	if r.Method == http.MethodGet && !owner {
		counted, err := s.store.CountView(r.Context(), slug)
		if err != nil {
			s.fail(w, r, err)
			return
		}
		if !counted { // another reader took the last view between the lookup and now
			s.gone(w, r)
			return
		}
		if v.MaxViews != nil {
			// This reader still sees the paste; it is removed once the limit is reached.
			defer s.store.DeleteIfBurned(r.Context(), slug)
		}
	}
	d.Data = readData{
		Paste: v, Body: template.HTML(v.BodyHTML), Gated: v.RevealMode == "tap",
		TimeLeft: paste.TimeLeft(v.ExpiresAt, s.now()), Date: v.CreatedAt.Format("2 January 2006"),
		URL: s.cfg.BaseURL + "/p/" + slug,
	}
	s.render(w, r, http.StatusOK, "read.html", d)
}
```

```go
	return ""
}

func trackProps(p store.Paste, expiry string) string {
	b, _ := json.Marshal(map[string]any{"slug": p.Slug, "reveal": p.RevealMode, "expiry": expiry, "view_limit": p.MaxViews != nil})
	return string(b)
}

// requireOwner loads the paste in the URL and continues only for its owner.
// Everyone else gets the gone page, the same as for a paste that does not exist.
func (s *Server) requireOwner(next func(http.ResponseWriter, *http.Request, store.Paste)) http.HandlerFunc {
	return s.requireUser(func(w http.ResponseWriter, r *http.Request) {
		slug := r.PathValue("slug")
		if !paste.ValidSlug(slug) {
			s.gone(w, r)
			return
		}
		p, err := s.store.PasteBySlug(r.Context(), slug)
		if err != nil {
			s.fail(w, r, err)
			return
		}
		if p.UserID != currentUser(r).ID {
			s.gone(w, r)
			return
		}
		next(w, r, p)
	})
}

func (s *Server) handleList(w http.ResponseWriter, r *http.Request) {
	pastes, err := s.store.ListPastesByUser(r.Context(), currentUser(r).ID, s.now())
	if err != nil {
		s.fail(w, r, err)
		return
	}
	rows := make([]listRow, 0, len(pastes))
	for _, p := range pastes {
		rows = append(rows, listRow{Paste: p, URL: s.cfg.BaseURL + "/p/" + p.Slug,
			Colour: paste.SignalColour(p.Slug), TimeLeft: paste.TimeLeft(p.ExpiresAt, s.now())})
	}
	s.render(w, r, http.StatusOK, "list.html", pageData{Title: "My pastes", Track: trackFrom(r), TrackProps: "{}", Data: rows})
}

func (s *Server) handleNewForm(w http.ResponseWriter, r *http.Request) {
	s.render(w, r, http.StatusOK, "form.html", pageData{Title: "New paste", Data: formData{Reveal: "tap", Expiry: "never"}})
}

// readForm validates the submitted paste fields onto p. The returned formData
// echoes what was typed, for re-rendering the form with a message.
func (s *Server) readForm(r *http.Request, p *store.Paste, editing bool) (formData, error) {
	fd := formData{Editing: editing, Slug: p.Slug, Title: r.PostFormValue("title"), Reveal: r.PostFormValue("reveal"),
		Expiry: r.PostFormValue("expiry"), MaxViews: r.PostFormValue("max_views")}
	raw := r.PostFormValue("body")
	if len(raw) > paste.MaxBodyBytes {
		return fd, errors.New("That's too long. Keep it under 100 KB.")
	}
	body := sanitize.HTML(raw)
	fd.Body = body

	title, err := paste.ValidateTitle(fd.Title)
	if err != nil {
		return fd, err
	}
	if sanitize.IsBlank(body) {
		return fd, errors.New("Write something first.")
	}
	if fd.Reveal != "immediate" && fd.Reveal != "tap" {
		return fd, errors.New("Pick how the text should appear.")
	}
	maxViews, err := paste.ParseMaxViews(fd.MaxViews)
	if err != nil {
		return fd, err
	}
	if !(editing && fd.Expiry == "keep") {
		exp, err := paste.ExpiresAt(fd.Expiry, s.now())
		if err != nil {
			return fd, err
		}
		p.ExpiresAt = exp
	}
	p.Title, p.BodyHTML, p.RevealMode, p.MaxViews, p.UpdatedAt = title, body, fd.Reveal, maxViews, s.now()
	return fd, nil
}

func (s *Server) handleCreate(w http.ResponseWriter, r *http.Request) {
	p := store.Paste{UserID: currentUser(r).ID, CreatedAt: s.now()}
	fd, err := s.readForm(r, &p, false)
	if err != nil {
		s.render(w, r, http.StatusUnprocessableEntity, "form.html", pageData{Title: "New paste", Error: err.Error(), Data: fd})
		return
	}
	// A slug collision is astronomically unlikely; retry a few times all the same.
	for attempt := 0; ; attempt++ {
		if p.Slug, err = paste.NewSlug(); err != nil {
			s.fail(w, r, err)
			return
		}
		err = s.store.CreatePaste(r.Context(), p)
		if !errors.Is(err, store.ErrConflict) || attempt == 4 {
			break
		}
	}
	if err != nil {
		s.fail(w, r, err)
		return
	}
	http.Redirect(w, r, "/p/"+p.Slug+"/ready?e=paste-create&x="+fd.Expiry, http.StatusSeeOther)
}

func (s *Server) handleReady(w http.ResponseWriter, r *http.Request, p store.Paste) {
	s.render(w, r, http.StatusOK, "ready.html", pageData{
		Title: "Link ready", Colour: paste.SignalColour(p.Slug),

		Data: readyData{Paste: p, URL: s.cfg.BaseURL + "/p/" + p.Slug},
	})
}

// expiryLabel keeps event properties to the known choices.
func expiryLabel(x string) string {
	switch x {
	case "1d", "1w", "1m", "never", "keep":
		return x
	}
	return ""
}

func (s *Server) handleEditForm(w http.ResponseWriter, r *http.Request, p store.Paste) {
	fd := formData{Editing: true, Slug: p.Slug, Title: p.Title, Body: p.BodyHTML, Reveal: p.RevealMode,
		Expiry: "keep", TimeLeft: paste.TimeLeft(p.ExpiresAt, s.now())}
	if p.MaxViews != nil {
		fd.MaxViews = strconv.Itoa(*p.MaxViews)
	}

}

func (s *Server) handleEdit(w http.ResponseWriter, r *http.Request, p store.Paste) {
	fd, err := s.readForm(r, &p, true)
	if err != nil {
		s.render(w, r, http.StatusUnprocessableEntity, "form.html",
			pageData{Title: "Edit paste", Colour: paste.SignalColour(p.Slug), Error: err.Error(), Data: fd})
		return

		s.fail(w, r, err)
		return
	}
	http.Redirect(w, r, "/p/"+p.Slug+"/ready?e=paste-edit&x="+fd.Expiry, http.StatusSeeOther)
}

func (s *Server) handleDelete(w http.ResponseWriter, r *http.Request, p store.Paste) {
	if err := s.store.DeletePaste(r.Context(), p.Slug, p.UserID); err != nil {
		s.fail(w, r, err)
		return
	}
	http.Redirect(w, r, "/?e=paste-delete", http.StatusSeeOther)
}
```

```go
package web

import (
	"net/http"
	"strconv"

	"github.com/santhoshjanan/pastr/internal/auth"
	"github.com/santhoshjanan/pastr/internal/store"
)

type adminData struct {
	SignupsOn bool
	Users     []store.User
}

func (s *Server) adminRoutes(mux *http.ServeMux) {
	mux.HandleFunc("GET /admin", s.requireAdmin(func(w http.ResponseWriter, r *http.Request) { s.renderAdmin(w, r, http.StatusOK, "") }))
	mux.HandleFunc("POST /admin", s.requireAdmin(s.handleAdmin))
}

func (s *Server) renderAdmin(w http.ResponseWriter, r *http.Request, status int, errMsg string) {
	on, err := s.store.SignupsEnabled(r.Context())
	if err != nil {
		s.fail(w, r, err)
		return
	}
	users, err := s.store.ListUsers(r.Context())
	if err != nil {
		s.fail(w, r, err)
		return
	}
	d := pageData{Title: "Admin", Error: errMsg, Data: adminData{SignupsOn: on, Users: users}}
	if r.URL.Query().Get("saved") == "1" && errMsg == "" {
		d.Notice = "Saved."
	}
	s.render(w, r, status, "admin.html", d)
}

func (s *Server) handleAdmin(w http.ResponseWriter, r *http.Request) {
	ctx, me := r.Context(), currentUser(r)
	reject := func(msg string) { s.renderAdmin(w, r, http.StatusUnprocessableEntity, msg) }

	action := r.PostFormValue("action")
	if action == "signups" {
		if err := s.store.SetSignupsEnabled(ctx, r.PostFormValue("on") == "true"); err != nil {
			s.fail(w, r, err)
			return
		}
		http.Redirect(w, r, "/admin?saved=1", http.StatusSeeOther)
		return
	}

	id, err := strconv.ParseInt(r.PostFormValue("user"), 10, 64)
	if err != nil {
		s.gone(w, r)
		return
	}
	target, err := s.store.UserByID(ctx, id)
	if err != nil {
		s.fail(w, r, err)
		return
	}
	switch action {
	case "disable":
		if target.ID == me.ID {
			reject("You can't disable your own account.")
			return
		}
		if target.IsAdmin {
			n, err := s.store.CountEnabledAdmins(ctx)
			if err != nil {
				s.fail(w, r, err)
				return
			}
			if n <= 1 {
				reject("That's the last admin. Someone has to hold the keys.")
				return
			}
		}
		err = s.store.SetDisabled(ctx, target.ID, true)
	case "enable":
		err = s.store.SetDisabled(ctx, target.ID, false)
	case "reset":
		pw := r.PostFormValue("password")
		if verr := auth.ValidatePassword(pw); verr != nil {
			reject(verr.Error())
			return
		}
		var hash string
		if hash, err = auth.HashPassword(pw); err == nil {
			if err = s.store.UpdatePassword(ctx, target.ID, hash); err == nil && target.ID != me.ID {
				err = s.store.DeleteUserSessions(ctx, target.ID)
			}
		}
	default:
		s.gone(w, r)
		return
	}
	if err != nil {
		s.fail(w, r, err)
		return
	}
	http.Redirect(w, r, "/admin?saved=1", http.StatusSeeOther)
}
```

```go
// Command pastr serves a self-hosted, spoiler-safe pastebin.
package main

import (
	"context"
	"errors"
	"fmt"
	"log/slog"
	"net"
	"net/http"
	"os"
	"os/signal"
	"path/filepath"
	"syscall"
	"time"

	"github.com/santhoshjanan/pastr/internal/auth"
	"github.com/santhoshjanan/pastr/internal/config"
	"github.com/santhoshjanan/pastr/internal/store"
	"github.com/santhoshjanan/pastr/internal/web"
)

func main() {
	log := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	if len(os.Args) > 1 && os.Args[1] == "healthcheck" {
		os.Exit(healthcheck())
	}
	if err := run(log); err != nil {
		log.Error("pastr stopped", "err", err)
		os.Exit(1)
	}
}

func run(log *slog.Logger) error {
	cfg, err := config.Load(os.Getenv)
	if err != nil {
		return err
	}
	if err := os.MkdirAll(filepath.Dir(cfg.DBPath), 0o700); err != nil {
		return err
	}
	st, err := store.Open(cfg.DBPath)
	if err != nil {
		return err
	}
	defer st.Close()

	ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
	defer stop()

	if err := bootstrapAdmin(ctx, cfg, st, time.Now()); err != nil {
		return err
	}
	srv, err := web.New(cfg, st, log)
	if err != nil {
		return err
	}
	go func() {
		tick := time.NewTicker(time.Hour)
		defer tick.Stop()
		for {
			sweep(ctx, st, time.Now(), log)
			select {
			case <-ctx.Done():
				return
			case <-tick.C:
			}
		}
	}()

	httpSrv := &http.Server{
		Addr: cfg.Addr, Handler: srv.Handler(),
		ReadHeaderTimeout: 5 * time.Second, ReadTimeout: 15 * time.Second,
		WriteTimeout: 30 * time.Second, IdleTimeout: 120 * time.Second,
		MaxHeaderBytes: 16 << 10,
	}
	errc := make(chan error, 1)
	go func() { errc <- httpSrv.ListenAndServe() }()
	log.Info("pastr listening", "addr", cfg.Addr, "base_url", cfg.BaseURL)

	select {
	case err := <-errc:
		return err
	case <-ctx.Done():
	}
	shutdownCtx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	defer cancel()
	if err := httpSrv.Shutdown(shutdownCtx); err != nil && !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	return nil
}

// bootstrapAdmin creates the first account from the environment when no users exist.
// Once any user exists the admin variables are ignored.
func bootstrapAdmin(ctx context.Context, cfg config.Config, st *store.Store, now time.Time) error {
	n, err := st.CountUsers(ctx)
	if err != nil || n > 0 {
		return err
	}
	if cfg.AdminUsername == "" {
		return errors.New("no users exist yet: set PASTR_ADMIN_USERNAME and PASTR_ADMIN_PASSWORD to create the first admin")
	}
	username, err := auth.ValidateUsername(cfg.AdminUsername)
	if err != nil {
		return fmt.Errorf("PASTR_ADMIN_USERNAME: %w", err)
	}
	if err := auth.ValidatePassword(cfg.AdminPassword); err != nil {
		return fmt.Errorf("PASTR_ADMIN_PASSWORD: %w", err)
	}
	hash, err := auth.HashPassword(cfg.AdminPassword)
	if err != nil {
		return err
	}
	_, err = st.CreateUser(ctx, username, hash, true, now)
	return err
}

func sweep(ctx context.Context, st *store.Store, now time.Time, log *slog.Logger) {
	pastes, err := st.DeleteExpiredPastes(ctx, now)
	if err != nil {
		log.Error("sweep pastes", "err", err)
	}
	sessions, err := st.DeleteExpiredSessions(ctx, now)
	if err != nil {
		log.Error("sweep sessions", "err", err)
	}
	if pastes+sessions > 0 {
		log.Info("swept", "pastes", pastes, "sessions", sessions)
	}
}

func healthURL(addr string) string {
	_, port, err := net.SplitHostPort(addr)
	if err != nil {
		port = "8080"
	}
	return "http://127.0.0.1:" + port + "/healthz"
}

// healthcheck is the container's HEALTHCHECK command; the image has no shell or curl.
func healthcheck() int {
	addr := os.Getenv("PASTR_ADDR")
	if addr == "" {
		addr = ":8080"
	}
	client := http.Client{Timeout: 3 * time.Second}
	resp, err := client.Get(healthURL(addr))
	if err != nil {
		return 1
	}
	resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return 1
	}
	return 0
}
```
